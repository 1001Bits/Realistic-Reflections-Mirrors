enable_testing()
find_path(MIRROR_CK_STB_INCLUDE stb_image.h REQUIRED)
add_library(MirrorCKAssets STATIC Nif.cpp Archive.cpp)
target_include_directories(MirrorCKAssets PRIVATE "${MIRROR_CK_STB_INCLUDE}" "${PROJECT_SOURCE_DIR}/src/Features")
add_library(MirrorsOfFalloutCK SHARED MirrorCK.cpp Picker.cpp)
add_executable(MirrorCKLauncher WIN32 Launcher.cpp)
set_target_properties(MirrorCKLauncher PROPERTIES OUTPUT_NAME MirrorsOfFalloutCK)
target_link_libraries(MirrorsOfFalloutCK PRIVATE MirrorCKAssets comctl32 bcrypt)
target_link_libraries(MirrorCKLauncher PRIVATE comdlg32 shell32 bcrypt)
foreach(target MirrorCKAssets MirrorsOfFalloutCK MirrorCKLauncher)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_options(${target} PRIVATE /utf-8)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
endforeach()
add_executable(mirror_ck_asset_tests tests/AssetTests.cpp)
target_compile_features(mirror_ck_asset_tests PRIVATE cxx_std_20)
target_link_libraries(mirror_ck_asset_tests PRIVATE MirrorCKAssets)
target_include_directories(mirror_ck_asset_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(MIRROR_CK_FIXTURES "${CMAKE_CURRENT_BINARY_DIR}/test-data")
add_custom_command(OUTPUT "${MIRROR_CK_FIXTURES}/ready.txt"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixtures.py" "${MIRROR_CK_FIXTURES}"
    DEPENDS tests/make_fixtures.py tests/fixtures/Flat.nif tests/fixtures/Hole.nif tests/fixtures/CurvedReject.nif)
add_custom_target(mirror_ck_fixtures DEPENDS "${MIRROR_CK_FIXTURES}/ready.txt")
add_dependencies(mirror_ck_asset_tests mirror_ck_fixtures)
add_test(NAME MirrorCKAssets COMMAND mirror_ck_asset_tests "${MIRROR_CK_FIXTURES}")
set_tests_properties(MirrorCKAssets PROPERTIES LABELS "Mirrors;Authoring" TIMEOUT 60)
